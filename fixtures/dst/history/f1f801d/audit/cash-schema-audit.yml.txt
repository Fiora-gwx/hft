name: Full cold ledger audit
on:
  push:
    branches: [codex/m2-cash-audit]
permissions:
  contents: read
jobs:
  audit:
    runs-on: ubuntu-latest
    timeout-minutes: 25
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - name: Free runner disk
        run: bash .github/scripts/free-disk.sh
      - name: Install pinned Rust toolchain
        run: bash .github/scripts/install-rust.sh
      - name: Prepare the standard workspace metadata
        run: |
          mkdir -p audit
          cp Cargo.lock audit/before.lock
          cargo xtask --help >audit/xtask.txt
          cp Cargo.lock audit/resolved.lock
          git diff -- Cargo.lock >audit/lock.patch
      - uses: Swatinem/rust-cache@v2
        with:
          key: x86-ci-v2
          save-if: false
      - name: Ledger observer and real snapshot boundaries
        run: |
          mkdir -p audit
          git rev-parse HEAD >audit/head.txt
          set +e
          cargo test --locked --profile ci -p hft-ledger --test ledger -- --nocapture >audit/recovery.txt 2>&1
          recovery_rc=$?
          cargo test --locked --profile ci -p hft-sequencer --test seq settlement:: -- --nocapture >audit/settlement.txt 2>&1
          settlement_rc=$?
          cargo test --locked --profile ci -p hft-event evt_075 -- --nocapture >audit/notes.txt 2>&1
          notes_rc=$?
          cargo test --locked --profile ci -p hft-testkit --features seq --lib led_267 -- --nocapture >audit/ledger.txt 2>&1
          ledger_rc=$?
          cargo test --locked --profile ci -p hft-testkit --features seq --test testkit snapshot_checks::tk_115 -- --nocapture >audit/boundaries.txt 2>&1
          boundary_rc=$?
          cargo xtask dst --hashtrail check --first 50 >audit/hashtrail.txt 2>&1
          trail_rc=$?
          cargo test --locked --profile ci -p hft-testkit --features seq --test testkit -- --nocapture >audit/testkit.txt 2>&1
          kit_rc=$?
          cp fixtures/dst/hashtrail.toml audit/hashtrail-before.toml
          cargo xtask dst --hashtrail write >audit/candidate.txt 2>&1
          candidate_rc=$?
          cp fixtures/dst/hashtrail.toml audit/hashtrail-candidate.toml
          cargo xtask gen-schema >audit/schema.txt 2>&1
          schema_rc=$?
          git diff --binary >audit/generated.patch
          printf 'recovery=%s settlement=%s notes=%s ledger=%s boundary=%s hashtrail=%s testkit=%s candidate=%s schema=%s\n' "$recovery_rc" "$settlement_rc" "$notes_rc" "$ledger_rc" "$boundary_rc" "$trail_rc" "$kit_rc" "$candidate_rc" "$schema_rc" | tee audit/status.txt
          cat audit/recovery.txt audit/settlement.txt audit/notes.txt audit/ledger.txt audit/boundaries.txt audit/hashtrail.txt audit/testkit.txt audit/candidate.txt audit/schema.txt
          test "$recovery_rc$settlement_rc$notes_rc$ledger_rc$boundary_rc$kit_rc$candidate_rc$schema_rc" = '00000000'
      - name: Attribute the complete HashTrail changes
        if: always()
        run: |
          cargo run --locked --profile ci -p hft-testkit --features seq --example balance-note-audit >audit/current.tsv
          cp crates/hft-ledger/src/cash.rs audit/cash-before.rs
          python3 - <<'PY'
          from pathlib import Path
          path = Path('crates/hft-ledger/src/cash.rs')
          source = path.read_text()
          start = source.index('        // The cold consumer has no OMS.')
          end = source.index('        verdict\n', start)
          assert source[start:end].count('code::BALANCE_CHECK') == 1
          path.write_text(source[:start] + source[end:])
          PY
          git diff -- crates/hft-ledger/src/cash.rs >audit/rollback.patch
          cargo run --locked --profile ci -p hft-testkit --features seq --example balance-note-audit >audit/rollback.tsv
          cp audit/cash-before.rs crates/hft-ledger/src/cash.rs
          python3 .github/scripts/balance-note-compare.py
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: full-cold-ledger-audit
          path: audit/
          retention-days: 7
